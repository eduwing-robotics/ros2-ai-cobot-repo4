cmake_minimum_required(VERSION 3.22)
project(fr5_motion_geometry)

find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(nlohmann_json REQUIRED)

add_executable(fr5_native_geometry src/native_geometry.cpp)
target_compile_features(fr5_native_geometry PRIVATE cxx_std_17)
target_compile_options(fr5_native_geometry PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(fr5_native_geometry moveit_core moveit_msgs rclcpp urdf srdfdom geometric_shapes)
target_link_libraries(fr5_native_geometry nlohmann_json::nlohmann_json)
install(TARGETS fr5_native_geometry DESTINATION lib/${PROJECT_NAME})
ament_package()
